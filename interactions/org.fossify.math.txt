# Fossify Calculator: 7 + 8 = shows 15.
# P2 conversion (element-addressed): the keypad keys are plain TextViews, so
# `find` by label replaces the 1080x1920-coordinate taps that broke whenever
# the layout shifted. The display result is asserted in the view tree
# (wait-find) instead of OCR.
wait 2
wait-find "7" timeout=40
tap $el
wait-find "+" timeout=10
tap $el
wait-find "8" timeout=10
tap $el
wait-find "=" timeout=10
tap $el
wait 1
shot result
expect-changed
wait-find "15" timeout=10
